\section{จงพิสูจน์ว่า $(a \mod n) = (b \mod n)$ แล้ว $n \mid (a-b)$}
}
\begin{align}
	let \ a &= nk_1 + p, p \in \mathbb{Z}_{\ge 0} \\
	let \ b &= nk_2 + p, p \in \mathbb{Z}_{\ge 0} \\
	a - b &= (nk_1 + p) - (nk_2 + p) \\ 
	a - b &= nk_1 + p - nk_2 - p \\
	a - b &= nk_1 - nk_2 \\
	a - b &= n(k_1 - k_2) \\
	let \ x &= k_1 - k_2 \\
	x &= \mathbb{Z}_{\ge 0} - \mathbb{Z}_{\ge 0} \\
	\therefore x &\in \mathbb{Z} \\
	a - b &= nx, x \in \mathbb{Z}
\end{align}
\subitem
$\therefore$ $(a \mod n) = (b \mod n)$ แล้ว $n \mid (a-b)$
