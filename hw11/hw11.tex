\section{จงพิสูจน์ว่า ตัวเลขที่หาร 3 ลงตัว จะมีผลรวมเลขโดดแต่ละตัว หาร 3 ลงตัวด้วย}
}
\begin{align}
	let \ x &= a_n \cdot 10^{n} + a_{n-1} \cdot 10^{n-1} + ... + a_1 \cdot 10^{1} + a_0 \\
	and \ 3 \mid x
\end{align}
\subsection{Prove by Direct Prove}
\begin{align}
	from\ 10^{m} \mod 3 &= 1, m \in \mathbb{Z}_{\geq0} \\
	x &\equiv a_n + a_{n-1} + ... + a_1 + a_0 \quad ( \mod 3) \\
	from \ 3 &\mid x \\
	x &\equiv 0 \quad ( \mod 3)\\
	0 &\equiv a_n + a_{n-1} + ... + a_1 + a_0 \quad ( \mod 3)
\end{align}
$\therefore$ ตัวเลขที่หาร 3 ลงตัว จะมีผลรวมเลขโดดแต่ละตัว หาร 3 ลงตัวด้วย
