{\small
\section{$Premises: \ p \implies ( q \land r), s \implies r,r  \implies p. Prove: s \implies q$}
}
\begin{align}
	&p \implies (q \land r) \quad & Premise \\
	&s \implies r \quad & Premise \\
	&r \implies p \quad & Premise \\
	&s \implies p \quad & 2,3 \ Hypothetical Syllogism \\
	&s \implies (q \land r) \quad & 4,1 \ Hypothetical Syllogism \\
	&\neg s \lor (q \land r) \quad & 5 \ Material Implication \\
	&(\neg s \lor q) \land (\neg s \lor r) \quad & 6 \ Distribution \\
	&\neg s \lor q \quad & 7 \ Simplication \\
	&s \implies q \quad & 8 \ Material Implication
\end{align}
\therefore {$p \implies ( q \land r), s \implies r,r  \implies p. Prove: s \implies q$}

\section{$Premises: \neg (r \lor s), \neg p \implies s, p \implies q. Prove: q$}
}
\begin{align}
	&\neg (r \lor s) \quad & Premise \\
	&\neg p \implies s \quad & Premise \\
	&p \implies q \quad & Premise \\
	&\neg s \implies p \quad & 2 \ Contrapositive \\
	&\neg s \implies q \quad & 3,4 \ Hypothetical Syllogism \\
	&\neg r \land \neg s \quad & 1 \ De Morgan \\
	&\neg s \quad & 6 \ Simplication \\
	&q \quad & 5,7 \ Modus Ponens
\end{align}
\therefore {$Premises: \neg (r \lor s), \neg p \implies s, p \implies q. Prove: q$}
