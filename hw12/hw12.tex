\section{จงหาเลขหลักหน่วยของ $7^{1665}$}
}
\begin{align}
	7^{1} \mod 10 &= 7 \mod 10 = 7 \\
	7^{2} \mod 10 &= 49 \mod 10 = 9 \\
	7^{3} \mod 10 &= 343 \mod 10 = 3 \\
	7^{4} \mod 10 &= 2401 \mod 10 = 1 \\
	7^{5} \mod 10 &= 16807 \mod 10 = 7 \\
	7^{6} \mod 10 &= 111649 \mod 10 = 9 \\
	7^{7} \mod 10 &= 823543 \mod 10 = 3 \\
	7^{8} \mod 10 &= 5764801 \mod 10 = 1 \\
	&...
\end{align}
\begin{center}
	this\ pattern\ repeat\ every\ 4\ step \\
\end{center}
\begin{align}
	\therefore 1665 \mod 4 &= 1 \\
	\therefore 7^{1665} \mod 10 &= 7^{1} \mod 10 = 7
\end{align}
\subitem
$\therefore$ เลขหลักหน่วยของ $7^{1665}$ คือ $7$
