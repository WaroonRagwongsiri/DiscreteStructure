{\small
\section{$(Q \lor R) \implies P \equiv (Q \implies P) \land (R \implies P)$}
}
\begin{align}
	(Q \lor R) \implies P &\equiv (Q \implies P) \land (R \implies P) \\
	\neg (Q \lor R) \lor P &\equiv (Q \implies P) \land (R \implies P), Material Implication \\
	(\neg Q \land \neg R) \lor P &\equiv (Q \implies P) \land (R \implies P), De Morgan \\
	(\neg Q \lor P) \land (\neg R \lor P)&\equiv (Q \implies P) \land (R \implies P), Distribution \\
	(Q \implies P) \land (R \implies P)&\equiv (Q \implies P) \land (R \implies P), Material Implication
\end{align}
\therefore $(Q \lor R) \implies P \equiv (Q \implies P) \land (R \implies P)$

{\small
\section{$((p \lor q) \land \neg p) \implies q$ เป็น Tautology}
}
\begin{align}
	((p \lor q) \land \neg p) \implies q &\equiv T \\
	\neg ((p \lor q) \land \neg p) \lor q &\equiv T, Material Implication \\
	(\neg (p \lor q) \lor p) \lor q &\equiv T, De Morgan \\
	\neg (p \lor q) \lor (p \lor q) &\equiv T, Association
\end{align}
\therefore $((p \lor q) \land \neg p) \implies q$ เป็น Tautology
